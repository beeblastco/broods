  Spec.Workers.Effect.refused x ∈ (Spec.Workers.step L s i).snd →
    s.phase x = Spec.Workers.Phase.absent ∧
      ∃ place, i = Spec.Workers.Input.dispatch x place ∧ ∀ (p : Spec.Workers.PlaceId), place = some p → ¬p ∈ s.places

Spec.Workers.slot_numbers_bounded
  ∀ {L : Spec.Workers.Limits},
  0 < L.maxSlots →
    ∀ (trace : List (Spec.Workers.system L).Input) (e : (Spec.Workers.system L).Effect),
      e ∈ (Spec.Workers.system L).effects trace →
        ∀ (x : Spec.Workers.RunId) (slot : Nat), e = Spec.Workers.Effect.started x slot → 1 ≤ slot ∧ slot ≤ L.maxSlots

Spec.Workers.slots_bounded
  ∀ {L : Spec.Workers.Limits}, 0 < L.maxSlots → Spec.Always (Spec.Workers.system L) fun s => s.active ≤ L.maxWorkers

Spec.Workers.start_effects
  ∀ {s : Spec.Workers.State} {r : Spec.Workers.RunId} {used : Nat} {e : Spec.Workers.Effect},
  e ∈ (Spec.Workers.start s r used).snd → e = Spec.Workers.Effect.started r (used + 1)

Spec.Workers.started_run_can_wait_again
  ((Spec.Workers.system Spec.Workers.yieldOnce).runFrom
          ((Spec.Workers.system Spec.Workers.yieldOnce).final
            [Spec.Workers.Input.dispatch 0 none, Spec.Workers.Input.dispatch 1 none,
              Spec.Workers.Input.dispatch 2 none])
          [Spec.Workers.Input.settle 0, Spec.Workers.Input.yieldSlot 1]).fst.phase
    1 =
  Spec.Workers.Phase.waiting 1

Spec.Workers.started_stays
  ∀ {L : Spec.Workers.Limits} (r : Spec.Workers.RunId) (u : Nat) (schedule : List Spec.Workers.Input)
  (s : Spec.Workers.State),
  Spec.Workers.started u (s.phase r) = true →
    Spec.Workers.started u (((Spec.Workers.system L).runFrom s schedule).fst.phase r) = true

Spec.Workers.started_step
  ∀ {L : Spec.Workers.Limits} (s : Spec.Workers.State) (i : Spec.Workers.Input) (r : Spec.Workers.RunId) (u : Nat),
  Spec.Workers.started u (s.phase r) = true → Spec.Workers.started u ((Spec.Workers.step L s i).fst.phase r) = true

Spec.Workers.waiting_step
  ∀ {L : Spec.Workers.Limits} {s : Spec.Workers.State} {r : Spec.Workers.RunId} {u k : Nat},
  Spec.Workers.Inv L s →
    ∀ (i : Spec.Workers.Input),
      r ∈ s.pending →
        List.idxOf r s.pending = k →
          s.phase r = Spec.Workers.Phase.waiting u →
            have after := (Spec.Workers.step L s i).fst;
            if Spec.Workers.Frees L s i = true then
              if k = 0 then Spec.Workers.started u (after.phase r) = true
              else r ∈ after.pending ∧ List.idxOf r after.pending = k - 1 ∧ after.phase r = Spec.Workers.Phase.waiting u
            else r ∈ after.pending ∧ List.idxOf r after.pending = k ∧ after.phase r = Spec.Workers.Phase.waiting u

Spec.Workers.work_conserving
  ∀ {L : Spec.Workers.Limits},
  0 < L.maxSlots → Spec.Always (Spec.Workers.system L) fun s => s.pending ≠ [] → s.active = L.maxWorkers
