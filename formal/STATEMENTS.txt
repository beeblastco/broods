Spec.Workers.enqueue.eq_1
  ∀ (L : Spec.Workers.Limits) (s : Spec.Workers.State) (r : Spec.Workers.RunId) (used : Nat) (loadShed : Bool),
  Spec.Workers.enqueue L s r used loadShed =
    if s.active < L.maxWorkers then Spec.Workers.start s r used
    else
      if (loadShed && Spec.Workers.full L s) = true then
        (s.setPhase r Spec.Workers.Phase.refused, [Spec.Workers.Effect.refused r])
      else
        (have __src := s.setPhase r (Spec.Workers.Phase.waiting used);
          { active := __src.active, pending := s.pending ++ [r], held := __src.held, running := __src.running,
            places := __src.places, phase := __src.phase },
          [])

Spec.Workers.release_moves_forward
  ∀ {L : Spec.Workers.Limits} {s : Spec.Workers.State} {x r : Spec.Workers.RunId} {n k : Nat},
  Spec.Workers.Inv L s →
    s.phase x = Spec.Workers.Phase.running n →
      r ∈ s.pending →
        List.idxOf r s.pending = k →
          have after := (Spec.Workers.release s x).fst;
          if k = 0 then
            (∀ (m : Nat), after.phase r ≠ Spec.Workers.Phase.waiting m) ∧ after.phase r ≠ Spec.Workers.Phase.absent
          else r ∈ after.pending ∧ List.idxOf r after.pending = k - 1

Spec.Workers.release_stays_out
  ∀ {L : Spec.Workers.Limits} {s : Spec.Workers.State} {x r : Spec.Workers.RunId} {n : Nat},
  Spec.Workers.Inv L s →
    s.phase x = Spec.Workers.Phase.running n →
      (∀ (m : Nat), s.phase r ≠ Spec.Workers.Phase.waiting m) →
        s.phase r ≠ Spec.Workers.Phase.absent →
          have after := (Spec.Workers.release s x).fst;
          (∀ (m : Nat), after.phase r ≠ Spec.Workers.Phase.waiting m) ∧ after.phase r ≠ Spec.Workers.Phase.absent

Spec.Workers.releasing_stays_out
  ∀ {L : Spec.Workers.Limits},
  0 < L.maxSlots →
    ∀ (r : Spec.Workers.RunId) (schedule : List Spec.Workers.Input) (s : Spec.Workers.State),
      Spec.Workers.Inv L s →
        Spec.Workers.Releasing L s schedule →
          (∀ (m : Nat), s.phase r ≠ Spec.Workers.Phase.waiting m) →
            s.phase r ≠ Spec.Workers.Phase.absent →
              (∀ (m : Nat), ((Spec.Workers.system L).runFrom s schedule).fst.phase r ≠ Spec.Workers.Phase.waiting m) ∧
                ((Spec.Workers.system L).runFrom s schedule).fst.phase r ≠ Spec.Workers.Phase.absent
